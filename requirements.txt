# =============================================================
#  STT Service — Python dependencies
#  Covers: main.py (STT service) + twilio_routes.py (API gateway)
#
#  Install order matters — torch must come from the CPU wheel index.
#  The Dockerfile handles that with a separate RUN step.
#  For plain pip installs run:
#
#    pip install torch==2.3.1 torchaudio==2.3.1 \
#        --index-url https://download.pytorch.org/whl/cpu
#    pip install -r requirements.txt
# =============================================================

# ── Web framework ─────────────────────────────────────────────
fastapi
uvicorn[standard]              # keep in sync with fastapi
websockets                     # required by uvicorn --ws websockets (Twilio stream)
python-multipart               # FastAPI File(...) / Form(...) uploads

# ── HTTP client (twilio_routes.py → LLM + TTS service calls) ──
httpx

# ── Twilio SDK (twilio_routes.py) ────────────────────────────
twilio

# ── STT — speech recognition ─────────────────────────────────
faster-whisper

# ── VAD — voice activity detection ───────────────────────────
# silero-vad loads via torch.hub; the pip package pins the weights.
silero-vad

# ── Audio processing ──────────────────────────────────────────
numpy                          # used directly in main.py (_ulaw_to_float32, etc.)
soundfile                      # WAV decoding in _audio_bytes_to_float32
# audioop is stdlib on Python ≤ 3.12 (used for µ-law decode + resample).
# audioop-lts is the backport for Python 3.13+ — installed conditionally
# in the Dockerfile, NOT listed here to avoid breaking Python 3.11/3.12.

# ── ML runtime ───────────────────────────────────────────────
# torch + torchaudio are installed separately from the CPU wheel index.
# Do NOT add them here — pip will pull the CUDA build from PyPI by default
# and balloon the image from ~800 MB to ~5 GB.

# ── Redis ─────────────────────────────────────────────────────
redis[asyncio]                 # aioredis.from_url used in both main.py and twilio_routes.py

# ── Logging ───────────────────────────────────────────────────
structlog
